// Q2. EMPLOYEE CRUD OPERATIONS

create 'Employee', 'details'

put 'Employee', '101', 'details:name', 'Amit'
put 'Employee', '101', 'details:salary', '30000'

get 'Employee', '101'

put 'Employee', '101', 'details:salary', '40000'

scan 'Employee'

delete 'Employee', '101', 'details:name'

deleteall 'Employee', '101'

disable 'Employee'
